Measurement::takeSample()
{
   // Will actually read temp sensors every 15s, so we skip
   // some samples.  When we read we also broadcast !
   // 660ms to get 2x sensors

   for all temp-sensors
      temperatureModule::readNextSensor()

   // Actually acquires power data each time
   for all power-sensors
      powerModule::readNextSensor()

   if > 30s since last stored sample then
      storageModule::storeSample( current sample )
}


storgeModule::storeSample
{
   for each sensor
      if emonCMS then
         send it
      endif

   if 5pm then
      send daily update
   endif

   // in saveSampleToBackingStore

   if 12pm then
      send daily readings
   endif

   if tboard then
      save to SPIFFS
   else
      save to SD
   endif
}
